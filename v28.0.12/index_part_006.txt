function renderGpsTrack(){
  try{
    if(typeof L==='undefined'||!mapObj)return;
    if(!gpsTrackLayer)gpsTrackLayer=L.layerGroup().addTo(mapObj);
    gpsTrackLayer.clearLayers();
    if(!gpsTrackEnabled()||!gpsLogbook||gpsLogbook.length<2)return;
    const pts=gpsLogbook.filter(p=>Number.isFinite(p.lat)&&Number.isFinite(p.lon)).map(p=>[p.lat,p.lon]);
    if(pts.length<2)return;
    L.polyline(pts,{color:'#2563eb',weight:4,opacity:.85}).addTo(gpsTrackLayer).bindPopup('Tatsächlich gesegelte Route<br>'+pts.length+' Logpunkte');
  }catch(e){console.warn('GPS-Track konnte nicht gezeichnet werden:',e)}
}
